====================================================
  FRIDAY BOT — COMMANDS (SIMPLE)
====================================================

PowerShell (Windows):
  .\setup.ps1              → one-time setup
  .\start_production.ps1   → kill all + open 5 windows (fresh start)
  .\restart_service.ps1    → restart ONE window only (listen/bot/live-*)

Do NOT commit: .env, telegram_session/, credentials JSON, *.lock, state.json

Do NOT start production twice (409 Conflict).
Do NOT run BOT from Cursor terminal.

  WHEN TO RESTART
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Edit JSON / Sheet / add-remove account  →  NO restart (LISTEN hot-reload ~45s)
  Change .py or .env code                 →  restart THAT service only
  Want all 5 windows fresh                →  start_production.ps1


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  GOOGLE SHEET COLUMNS (tab "Telegram Acc")
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Status (dropdown)
    · STAFF sets Pending  → triggers LIVE-LOGIN (login baru)
    · SCRIPT writes Ready / Banned / Frozen / Expired / Disconnected
    · SCRIPT never writes Pending or Pending Login

  Session Friday  (staff reference — script only)
    · Active          = bot session OK
    · Pending Login   = belum ada / belum valid session
    · Banned          = akun banned (mirror Status)
    · Frozen / Expired / Disconnected = same meaning

  OTP   = staff isi 5 digit saat login
  2FA   = cloud password jika ada


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  NEW LOGIN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1) Row di Sheet (Phone, Group, Acc Name, Device ID)
  2) Status = Pending  (staff only)
  3) LIVE-LOGIN SMS → isi OTP
  4) Status=Ready, Session Friday=Active


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  REPLACE BANNED (same row)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1) Banned → auto purge JSON + session (TELEGRAM_PURGE_BANNED_ACCOUNTS=1)
  2) Clear row → isi detail akun baru (phone baru OK)
  3) Status = Pending → login seperti akun baru
